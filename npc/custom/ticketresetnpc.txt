akasenzai,167,143,5	script	Reset Girl	124,{
	if (countitem(7859) < 1) {
		mes "[Reset Girl]";
		mes "I can reset your stats or skills for a <ITEM>Game Ticket<INFO>7859</INFO></ITEM>.";
		close;
	}

	mes "[Reset Girl]";
	mes "What would you like me to reset for a <ITEM>Game Ticket<INFO>7859</INFO></ITEM>?";

	set .@i,(select("Nothing:^FF3355Stats:Skills^000000"));
	if(.@i <= 1) close;
	
	delitem 7859,1;
	if (.@i == 2) {
		ResetStatus;
	} else if (.@i == 3) {
		ResetSkill;
	}

	mes "Done!";
	close;
}
